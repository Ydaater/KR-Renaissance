	GER_operation_alpenrose = {
		icon = GFX_decision_hol_draw_up_staff_plans

		allowed = { always = no } #activated by effect

		available = {
			if = {
				limit = { NOT = { has_country_flag = GER_first_OHL_operation } }
				custom_trigger_tooltip = {
					tooltip = GER_OHL_operation_lock_tt
					has_country_flag = {
						flag = GER_OHL_operation_lock
						days > 150
					}
				}
			}
			OR = {
			    has_war_with = SRI
			    has_war_with = SIC
			    has_war_with = SRD
			    has_war_with = ITA
			}
		}
		visible = {
			OR = {
				SRI = {
					exists = yes 
					has_war_with = ROOT
				}
				ITA = {
					exists = yes 
					has_war_with = ROOT
				}
				SRD = {
					exists = yes 
					has_war_with = ROOT
				}
				SIC = {
					exists = yes 
					has_war_with = ROOT
				}
			}
		}
		custom_cost_text = CP_50
		custom_cost_trigger = {
			NOT = { command_power < 50 }
		}

		fire_only_once = yes

		days_remove = 90
		complete_effect = {
			if = {
				limit = { has_country_flag = GER_first_OHL_operation } 
				clr_country_flag = GER_first_OHL_operation
			}
			set_country_flag = {
			    flag = GER_OHL_operation_lock
			    days = 0
			}
			add_command_power = -50
			if = {
				limit = { has_war_with = SRI }
				every_allied_country = {
					add_timed_idea = {
						idea = GER_operation_alpenrose_SRI_idea
						days = 90
					}
				}
			}
			else_if = {
				limit = { has_war_with = ITA }
				every_allied_country = {
					add_timed_idea = {
						idea = GER_operation_alpenrose_ITA_idea
						days = 90
					}
				}
			}
			else_if = {
				limit = { has_war_with = SIC }
				every_allied_country = {
					add_timed_idea = {
						idea = GER_operation_alpenrose_SIC_idea
						days = 90
					}
				}
			}
			else_if = {
				limit = { has_war_with = SRD }
				every_allied_country = {
					add_timed_idea = {
						idea = GER_operation_alpenrose_SRD_idea
						days = 90
					}
				}
			}
		}
		modifier = {
			command_power_gain = -0.1
		}

		ai_will_do = { base = 0 }
		fixed_random_seed = no
	}
